async function loadGallery() { try { const response = await fetch('./gallery-data.json'); const photos = await response.json(); const galleryElement = document.getElementById('gallery'); photos.forEach(photo => { const card = document.createElement('div'); card.className = 'photo-card'; const wrapper = document.createElement('div'); wrapper.className = 'image-wrapper'; const img = document.createElement('img'); // Encode spaces safely for browser fetching img.src = encodeURI(photo.src); img.alt = photo.filename; img.loading = "lazy"; const caption = document.createElement('div'); caption.className = 'caption'; caption.textContent = photo.filename; wrapper.appendChild(img); card.appendChild(wrapper); card.appendChild(caption); galleryElement.appendChild(card); }); } catch (err) { console.error("Failed to load images:", err); } } loadGallery();
